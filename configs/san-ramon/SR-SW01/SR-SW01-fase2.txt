! VERIFICAR:

!   show interfaces trunk -> trunks con native 100
!   ping 172.27.86.241 -> exito (gateway VLAN 900)
! SI FALLA:
!   NATIVE_VLAN_MISMATCH -> la nativa no es 100 en el otro extremo
!   PC con IP 169.254.x.x -> DHCP no llega: revisar trunks y subinterfaz del router
!============================================================
enable
configure terminal
hostname SR-SW01
no ip domain-lookup
!
! --- VLANs de la sede (mismos IDs en todas las sedes, D2.3) ---
vlan 100
 name GERENCIA
exit
vlan 300
 name VENTAS
exit
vlan 500
 name BODEGA
exit
vlan 600
 name CONTABILIDAD
exit
vlan 900
 name ADMIN-SSH
exit
!
! --- Puertos de acceso ---
interface range FastEthernet0/1-20
 description ACCESO-BODEGA
 switchport mode access
 switchport access vlan 500
!
interface range FastEthernet0/21-23
 description SIN-USO-PENDIENTE-SHUTDOWN-FASE5
!
interface FastEthernet0/24
 description ACCESO-ADMIN-LAPTOP
 switchport mode access
 switchport access vlan 900
!
! --- Troncales (nativa 100 en ambos extremos) ---
interface GigabitEthernet0/1
 description TRUNK-A-SR-RT01
 switchport mode trunk
 switchport trunk native vlan 100
!
interface GigabitEthernet0/2
 description TRUNK-A-SR-SW02
 switchport mode trunk
 switchport trunk native vlan 100
!
! --- Gestion del switch en VLAN 900 (D3.2) ---
interface Vlan900
 description MGMT-VLAN900
 ip address 172.27.86.242 255.255.255.248
 no shutdown
!
ip default-gateway 172.27.86.241
!
end
write memory

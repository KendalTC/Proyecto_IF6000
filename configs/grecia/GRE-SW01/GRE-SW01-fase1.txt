enable
configure terminal
hostname GRE-SW01

vlan 100
 name GERENCIA
exit
vlan 200
 name PRODUCCION
exit
vlan 300
 name VENTAS
exit
vlan 400
 name DESARROLLO
exit
vlan 900
 name ADMIN-SSH
exit

interface range FastEthernet0/1-22
 description ACCESO-VENTAS
 switchport mode access
 switchport access vlan 300
!
interface FastEthernet0/23
 description SIN-USO-PENDIENTE-SHUTDOWN-FASE5
!
interface FastEthernet0/24
 description ACCESO-ADMIN-LAPTOP
 switchport mode access
 switchport access vlan 900
!
interface GigabitEthernet0/1
 description TRUNK-A-GRE-RT01
 switchport mode trunk
 switchport trunk native vlan 100
!
interface GigabitEthernet0/2
 description TRUNK-A-GRE-SW02
 switchport mode trunk
 switchport trunk native vlan 100
!
interface Vlan900
 description MGMT-VLAN900
 ip address 172.27.86.114 255.255.255.248
 no shutdown
!
ip default-gateway 172.27.86.113

end
write memory

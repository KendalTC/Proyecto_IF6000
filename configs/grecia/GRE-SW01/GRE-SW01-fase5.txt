GRE-SW01>enable 
GRE-SW01#configure terminal 


GRE-SW01(config)#no ip domain-lookup
GRE-SW01(config)#ip domain-name clavitoveloz.com
GRE-SW01(config)#enable secret proyecto
GRE-SW01(config)#service password-encryption 
GRE-SW01(config)#username kendal privilege 15 secret redes
GRE-SW01(config)#username mariangel privilege 15 secret redes
GRE-SW01(config)#username sebas privilege 15 secret redes

GRE-SW01(config)#banner motd "ACCESO RESTRINGIDO, SOLO PERSONAL AUTORIZADO - EL CLAVITO VELOZ"

GRE-SW01(config)#line console 0
GRE-SW01(config-line)#login local
GRE-SW01(config-line)#logging synchronous
GRE-SW01(config-line)#exec-timeout 5 0
GRE-SW01(config-line)#exit

GRE-SW01(config)#line vty 0 15
GRE-SW01(config-line)#login local 
GRE-SW01(config-line)#transport input ssh
GRE-SW01(config-line)#exec-timeout 5 0
GRE-SW01(config-line)#exit
